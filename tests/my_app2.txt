import { Injectable } from '@angular/core';

@Injectable({ providedIn: 'root' })
export class ProductService {
  private storageKey = 'products';
  products: any[] = [];
  nextId = 1;

  constructor() {
     let stored = localStorage.getItem(this.storageKey);
    this.products = stored ? JSON.parse(stored) : [
      { id: 1, name: 'Product 1', image: '../images/1.jpg', details: 'School book ' },
      { id: 2, name: 'Product2', image: '../images/شنتة.jpg', details: 'bag for Laptop ' },
      { id: 3, name: 'Product 3', image: '../images/مقلمة.jpg', details: 'School tools ' }
    ];
    this.nextId = Math.max(0, ...this.products.map(products => products.id)) + 1;
    this.save   Products();
  }

    saveProducts() {
    localStorage.setItem(this.storageKey, JSON.stringify(this.products));
  }

  getAll() {
    return this.products;
  }

  getById(id: number) {
    return this.products.find(products => products.id === id);
  }

  add(product: any) {
    product.id = this.nextId++;
    this.products.push(product);
    this.saveProducts();
  }

  update(id: number, updated: any) {
    const index = this.products.findIndex(products => products.id === id);
    if (index > -1) {
      this.products[index] = { ...updated, id };
      this.saveProducts();
    }
  }

  deleteProduct(id: number) {
    this.products = this.products.filter(products => products.id !== id);
    this.saveProducts();
  }
}

import { Component } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';
import { CommonModule } from '@angular/common';
import { ProductService } from '../services/product.service';


@Component({
  standalone: true,
  imports: [CommonModule, FormsModule],
  template: `
    <div class="form-container">
    <h2>{{ isEdit ? ' edit ' : 'add' }} Information</h2>
    <form>
      <label>Name:
        <input [(ngModel)]="product.name" name="name" required />
      </label><br />
      <label>Link:
        <input [(ngModel)]="product.image" name="image" required />
      </label><br />
      <label>Information:
        <textarea [(ngModel)]="product.details" name="details" required></textarea>
      </label><br />
      <button type="button" (click)="save()">saved</button>
    </form>
  </div>
  `,
  styles: [`
    .form-container { padding: 20px; direction: rtl; }
    label { display: block; margin: 10px 0; }
    input, textarea { width: 100%; }
  `]
})
export class ProductFormComponent {
  product = { name: '', image: '', details: '' };
  isEdit = false;
  id: number = 0;

  constructor(private route: ActivatedRoute, private router: Router, private productService: ProductService) {
    let paramId = this.route.snapshot.paramMap.get('id');
    if (paramId) {
      this.isEdit = true;
      this.id = +paramId;
      const existing = this.productService.getById(this.id);
      if (existing) this.product = { ...existing };
    }
  }

  save() {
    if (this.isEdit) {
      this.productService.update(this.id, this.product);
    } else {
      this.productService.add(this.product);
    }
    this.router.navigate(['/']);
  }
}
